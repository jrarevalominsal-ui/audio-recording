Google Apps Script (envío a ar17071@ues.edu.sv + guardado de audio en Drive) 
const TEACHER_EMAIL = "ar17071@ues.edu.sv";
const DRIVE_FOLDER_NAME = "Speaking_Audio_Submissions";

function getOrCreateFolder(name) {
  const folders = DriveApp.getFoldersByName(name);
  if (folders.hasNext()) return folders.next();
  return DriveApp.createFolder(name);
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const folder = getOrCreateFolder(DRIVE_FOLDER_NAME);

    let audioBlob = null;
    let audioFileUrl = "No se adjuntó audio";

    if (data.audioBase64) {
      const decodedBytes = Utilities.base64Decode(data.audioBase64);
      const mimeType = data.audioMimeType || "audio/webm";
      const ext = mimeType.includes("mp4") ? "mp4" : "webm";
      const safeName = (data.studentName || "estudiante").replace(/[^a-zA-Z0-9_-]/g, "_");
      const fileName = `${safeName}_${data.examDate || ""}_${Date.now()}.${ext}`;

      audioBlob = Utilities.newBlob(decodedBytes, mimeType, fileName);
      const file = folder.createFile(audioBlob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      audioFileUrl = file.getUrl();
    }

    const subject = `Speaking (${data.finalGrade}/10.0) - ${data.studentName} (${data.studentCourse || "Sin curso"})`;
    const body = [
      `Estudiante: ${data.studentName}`,
      `Curso: ${data.studentCourse || "No especificado"}`,
      `Fecha: ${data.examDate || ""}`,
      `Calificación: ${data.finalGrade} / 10.0`,
      "",
      "--- Desglose ---",
      `• Tiempo: ${data.timeScore} / 2.5 (${data.durationSec}s)`,
      `• Fluidez: ${data.fluencyScore} / 2.5 (${data.wordCount} palabras, ~${data.wpm} WPM)`,
      `• Gramática (Presente/Pasado/Futuro): ${data.grammarScore} / 2.5 (${data.tensesCovered}/3 tiempos)`,
      `• Pronunciación: ${data.pronunciationScore} / 2.5 (${data.pronunciationPct}%)`,
      "",
      "--- Transcripción ---",
      data.transcript || "(Sin transcripción)",
      "",
      "--- Observaciones ---",
      data.observations || "Sin observaciones.",
      "",
      `Audio en Google Drive: ${audioFileUrl}`
    ].join("\n");

    const mailOptions = { to: TEACHER_EMAIL, subject, body };
    if (audioBlob) mailOptions.attachments = [audioBlob];
    MailApp.sendEmail(mailOptions);

    return ContentService
      .createTextOutput(JSON.stringify({ status: "ok", audioUrl: audioFileUrl }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
